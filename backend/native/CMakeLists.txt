cmake_minimum_required(VERSION 3.25)
project(busydata_native_transport LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME official_client_transport_policy COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_transport_policy.py)
if(WIN32)
  add_executable(busydata_tool_launcher tool_launcher_windows.c)
  target_link_libraries(busydata_tool_launcher PRIVATE ws2_32 advapi32)
  add_executable(busydata_pipe_transport_test pipe_transport_windows_test.c)
  add_executable(busydata_tool_launcher_test tool_launcher_windows_test.c)
  target_link_libraries(busydata_pipe_transport_test PRIVATE ws2_32)
  foreach(target busydata_tool_launcher busydata_pipe_transport_test busydata_tool_launcher_test)
    target_compile_options(${target} PRIVATE /W4 /utf-8)
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
  endforeach()
  add_test(NAME private_pipe_client_lifecycle COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test_windows_transport.py $<TARGET_FILE:busydata_tool_launcher> $<TARGET_FILE:busydata_pipe_transport_test> $<TARGET_FILE:busydata_tool_launcher_test>)
  set_tests_properties(private_pipe_client_lifecycle PROPERTIES TIMEOUT 180)
endif()
